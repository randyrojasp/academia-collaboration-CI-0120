# Salto hacia atrás y escritura en x0
jal x5, 12           # x5 = 4, salta a 0x0C
addi x1, x0, 7       # x1 = 7 después del salto hacia atrás
jal x0, 12           # x0 = 0, salta a 0x14
jal x6, -8           # x6 = 16, salta a 0x04
addi x2, x0, 99      # No se ejecuta: x2 = 0
addi x3, x5, 1       # x3 = 5
addi x4, x6, 1       # x4 = 17

ebreak               # pc = 0x1C
