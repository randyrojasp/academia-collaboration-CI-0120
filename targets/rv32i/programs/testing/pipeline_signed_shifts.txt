# Desplazamientos con números negativos
addi x1, x0, -1      # x1 = -1
addi x2, x0, -16     # x2 = -16
addi x3, x0, 2       # x3 = 2
sra x4, x2, x3       # x4 = -4
srl x5, x2, x3       # x5 = 0x3FFFFFFC
srai x6, x2, 2       # x6 = -4
srli x7, x2, 2       # x7 = 0x3FFFFFFC
slli x8, x1, 31      # x8 = 0x80000000

# El desplazamiento usa los 5 bits bajos del registro
addi x9, x0, 34      # x9 = 34
srl x10, x8, x9      # x10 = 0x20000000
sra x11, x8, x9      # x11 = 0xE0000000

# Comparaciones con signo y sin signo
slt x12, x1, x0      # x12 = 1
sltu x13, x1, x0     # x13 = 0
slti x14, x2, -1     # x14 = 1
sltiu x15, x1, -1    # x15 = 0

# Límites del inmediato y registro x0
addi x16, x0, -2048  # x16 = -2048
addi x17, x0, 2047   # x17 = 2047
add x18, x16, x17    # x18 = -1
addi x0, x0, 123     # x0 = 0
addi x19, x0, 1      # x19 = 1
sll x20, x3, x9      # x20 = 8
sll x21, x3, x0      # x21 = 2

ebreak               # pc = 0x58
