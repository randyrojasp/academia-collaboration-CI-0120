# Desplazamientos usando registros
addi x1, x0, 16     # x1 = 16
addi x2, x0, 2      # x2 = 2
sll x3, x1, x2      # x3 = 64
srl x4, x1, x2      # x4 = 4
sra x5, x1, x2      # x5 = 4

# Desplazamientos usando inmediatos
slli x6, x1, 2      # x6 = 64
srli x7, x1, 2      # x7 = 4
srai x8, x1, 2      # x8 = 4

# Desplazamientos con números negativos
addi x9, x0, -16    # x9 = -16
sra x10, x9, x2     # x10 = -4
srai x11, x9, 2     # x11 = -4
srl x12, x9, x2     # x12 = 1073741820 (RV32)
