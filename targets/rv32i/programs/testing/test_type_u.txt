# LUI
lui x1, 0x12345         # x1 = 0x12345000
lui x2, 0x00001         # x2 = 0x00001000
lui x3, 0xFFFFF         # x3 = 0xFFFFF000
lui x4, 0x00000         # x4 = 0x00000000

# AUIPC
auipc x5, 0x00001       # x5 = PC + 0x00001000
auipc x6, 0x00000       # x6 = PC
auipc x7, 0x00002       # x7 = PC + 0x00002000

# LUI con ADDI
lui x8, 0x12345         # x8 = 0x12345000
addi x8, x8, 0x678      # x8 = 0x12345678

# LUI con negativo
lui x9, 0xFFFFF         # x9 = 0xFFFFF000
addi x9, x9, -1         # x9 = 0xFFFFEFFF

ebreak
